cmake_minimum_required(VERSION 3.16)
project(libhddvd VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(HDDVD_ENABLE_REFERENCE_TESTS "Enable tests using an external reference ACA" OFF)
option(HDDVD_BUILD_FUZZERS "Build libFuzzer harnesses (Clang only)" OFF)
find_package(LibXml2 REQUIRED)
add_library(hddvd src/hddvd.c src/io.c src/util.c src/xpl.c src/xmf.c src/aca.c src/disc.c src/ifo.c src/map.c src/evo.c)
add_library(hddvd::hddvd ALIAS hddvd)
target_compile_features(hddvd PUBLIC c_std_99)
target_include_directories(hddvd PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(hddvd PRIVATE LibXml2::LibXml2)
set_target_properties(hddvd PROPERTIES C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
if(MSVC)
  target_compile_options(hddvd PRIVATE /W4)
else()
  target_compile_options(hddvd PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(hddvd-info tools/hddvd-info.c)
target_link_libraries(hddvd-info PRIVATE hddvd)
add_executable(hddvd-menuinfo tools/hddvd-menuinfo.c)
target_link_libraries(hddvd-menuinfo PRIVATE hddvd)
add_executable(hddvd-acainfo tools/hddvd-acainfo.c)
target_link_libraries(hddvd-acainfo PRIVATE hddvd)
add_executable(hddvd-acaextract tools/hddvd-acaextract.c)
target_link_libraries(hddvd-acaextract PRIVATE hddvd)
if(BUILD_TESTING)
 add_executable(test-core tests/test_core.c)
 target_link_libraries(test-core PRIVATE hddvd)
 add_test(NAME core COMMAND test-core)
 add_executable(test-xpl tests/test_xpl.c)
 target_link_libraries(test-xpl PRIVATE hddvd)
 add_test(NAME xpl COMMAND test-xpl)
 add_executable(test-xmf tests/test_xmf.c)
 target_link_libraries(test-xmf PRIVATE hddvd)
 target_compile_definitions(test-xmf PRIVATE
  HDDVD_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
 add_test(NAME xmf COMMAND test-xmf)
 if(HDDVD_ENABLE_REFERENCE_TESTS)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
  add_test(NAME reference-menu
   COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_reference_menu.py
    $<TARGET_FILE:hddvd-acainfo>
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_menu_expected.json)
  set_tests_properties(reference-menu PROPERTIES SKIP_RETURN_CODE 77)
 endif()
endif()
install(TARGETS hddvd hddvd-info hddvd-menuinfo hddvd-acainfo hddvd-acaextract EXPORT hddvdTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/hddvd.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT hddvdTargets NAMESPACE hddvd:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/hddvd)
if(BUILD_TESTING)
 add_executable(test-evo tests/test_evo.c)
 target_link_libraries(test-evo PRIVATE hddvd)
 add_test(NAME evo COMMAND test-evo)
endif()
if(HDDVD_BUILD_FUZZERS)
 if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "HDDVD_BUILD_FUZZERS requires Clang and libFuzzer")
 endif()
 foreach(target aca xmf xmu)
  add_executable(fuzz-${target} tests/fuzz_${target}.c)
  target_link_libraries(fuzz-${target} PRIVATE hddvd)
  target_compile_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
 endforeach()
endif()
if(BUILD_TESTING)
 add_executable(test-aca tests/test_aca.c)
 target_link_libraries(test-aca PRIVATE hddvd)
 add_test(NAME aca COMMAND test-aca)
endif()
